const { jsPDF } = window.jspdf; let shipments = []; document.getElementById('shipping-form').addEventListener('submit', (e) => { e.preventDefault(); const shipment = { id: Date.now(), sender: document.getElementById('sender').value, receiver: document.getElementById('receiver').value, address: document.getElementById('address').value, phone: document.getElementById('phone').value, state: document.getElementById('state').value, items: document.getElementById('items').value, weight: parseFloat(document.getElementById('weight').value) }; shipments.push(shipment); renderDashboard(); e.target.reset(); }); function renderDashboard() { const container = document.getElementById('states-container'); container.innerHTML = ''; // Agrupar por estado const grouped = shipments.reduce((acc, s) => { if (!acc[s.state]) acc[s.state] = []; acc[s.state].push(s); return acc; }, {}); for (let state in grouped) { const stateDiv = document.createElement('div'); stateDiv.innerHTML = `

${state} (${grouped[state].length})

`; grouped[state].forEach(s => { stateDiv.innerHTML += `
Para: ${s.receiver} | Peso: ${s.weight} lbs
`; }); container.appendChild(stateDiv); } } document.getElementById('generate-pdf').addEventListener('click', () => { const doc = new jsPDF(); let yPos = 10; shipments.forEach((s, index) => { // Lógica para 2 fichas por hoja if (index > 0 && index % 2 === 0) { doc.addPage(); yPos = 10; } else if (index % 2 !== 0) { yPos = 150; // Segunda ficha en la mitad de la hoja } doc.setDrawColor(0); doc.rect(10, yPos, 190, 130); // Recuadro de la ficha doc.setFontSize(16); doc.text(`SANDRA EXPRESS - Guía #${s.id}`, 15, yPos + 10); doc.setFontSize(12); doc.text(`Remitente: ${s.sender}`, 15, yPos + 30); doc.text(`Destinatario: ${s.receiver}`, 15, yPos + 40); doc.text(`Estado: ${s.state}`, 15, yPos + 50); doc.text(`Dirección: ${s.address}`, 15, yPos + 60); doc.text(`Artículos: ${s.items}`, 15, yPos + 80); doc.text(`Peso Total: ${s.weight} lbs`, 15, yPos + 110); }); doc.save('envios-sandra-express.pdf'); });
